cmake_minimum_required(VERSION 3.16)
if(MIKU_HOST)
  project(MikuOS.Host LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  find_package(Threads REQUIRED)
  add_executable(miku-host platform/host.cpp kernel/kernel.cpp shell/shell.cpp protocol/protocol.cpp)
  target_include_directories(miku-host PRIVATE .)
  target_link_libraries(miku-host PRIVATE Threads::Threads)
  enable_testing()
  add_executable(kernel-tests ../tests/kernel_tests.cpp kernel/kernel.cpp shell/shell.cpp protocol/protocol.cpp)
  target_include_directories(kernel-tests PRIVATE .)
  add_test(NAME kernel COMMAND kernel-tests)
else()
  include($ENV{IDF_PATH}/tools/cmake/project.cmake)
  project(MikuOS.Firmware)
endif()
